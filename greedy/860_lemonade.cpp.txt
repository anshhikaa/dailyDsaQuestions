//problem : lemonate change 
// platform : leetcode 860
// approach track 5 and 10 
// tc = o(n) , o(1)
class Solution {
public:
    bool lemonadeChange(vector<int>& bills) {
        int ten = 0 ;
        int five = 0 ;
        for(int i = 0 ;i < bills.size();i++){
            if(bills[i] == 5){
                five++;
            }
            else if(bills[i] == 10){
                if(five == 0) return false;
                ten ++;
                five --;
            }
            else { 
                // 20 (15= 10,5 or 5,5,5)
                if(ten > 0 && five > 0){
                    ten -- ;
                    five --;
                }
                else if(five >= 3){
                    five -= 3 ;
                }
                else{
                    return false;
                }
            }
        }
      return true;  
    }
};
