//problem : jump game 1 leetcode 55 
// approach - bfs - tc = o(n^2), sc = o(n)
class Solution {
public:
    bool canJump(vector<int>& nums) {
        queue<pair<int,int>> q ; //->{max jump , index}
        int n = nums.size();
        vector<int>vis(n,0);
        q.push({nums[0],0});
        vis[0] = 1 ;
        int lastIndex = nums.size()-1;
       while(!q.empty()){
        auto[maxJump,index] = q.front();
        q.pop();
        int jumpIndex = index + maxJump;
        if(jumpIndex >= lastIndex ){
            return true;
        }
        else {
            for(int i = index+1 ; i <= index + maxJump && i < n ; i++){
                if(!vis[i]){
                q.push({nums[i],i});
                vis[i] = 1;
                }
            }
        }
       }
      return false;
    }
};

// optimised version - greedy 
class Solution {
public:
    bool canJump(vector<int>& nums) {
        int maxReach = 0 ;
        for(int i = 0 ; i< nums.size();i++){
            if(i > maxReach) return false;
            maxReach = max(maxReach,i+nums[i]);
        }
     return true; 
    }
};
