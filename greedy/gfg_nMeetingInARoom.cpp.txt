//problem : n meeeting in a room 
// platform : gfg 
// approach - combine start & end in one vector<pair> , sort by end time , compare start time w end time 
// tc  =o(nlogn), sc = o(n)
class Solution {
  public:
    // Function to find the maximum number of meetings that can
    // be performed in a meeting room.
    int maxMeetings(vector<int>& start, vector<int>& end) {
        vector<pair<pair<int,int>,int>>meeting;
        for(int i = 0 ; i< start.size(); i++){
            meeting.push_back({{start[i],end[i]},i+1});
        }
        //sorting
        sort(meeting.begin(), meeting.end(), [](auto &a, auto &b){
    return a.first.second < b.first.second;});
        //greedy comparison 
        int count = 1 ;
        if(meeting.size()==0) return 1 ;
        int freeTime = meeting[0].first.second ;
        for(int i = 1 ; i<meeting.size();i++){
           int startTime = meeting[i].first.first;
           if(startTime>freeTime){
               count ++;
               freeTime = meeting[i].first.second;
           }
        }
        return count ;
    }
};
