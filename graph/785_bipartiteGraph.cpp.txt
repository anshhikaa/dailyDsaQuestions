//problem : bipartite graph
// leetcode : 785
//using bfs + multipleComponent tc = o(v+e) , sc = o(v)
class Solution {
public:
bool multipleComponent(int startNode , int v ,vector<vector<int>>& graph,int colour[]){
     queue<int>q ; 
        q.push(startNode);
          colour[startNode] = 0 ;
        while(!q.empty()){
            int node = q.front();
            q.pop();
            for(auto it : graph[node]){
                // not coloured then opposite colour
                if(colour[it] == -1){
                    colour[it] = !colour[node];
                    q.push(it);
                }
                //already colourd then check if bipartite condition violated
                else if(colour[it] == colour[node]){
                    return false; 
                }
            }
        }
        return true;
}
    bool isBipartite(vector<vector<int>>& graph) {
        int v = graph.size();
        int colour[v];
        for(int i = 0 ; i < v ; i++){
            colour[i] = -1;
        }
        for(int i = 0 ; i< v ;i++){
            if(colour[i] == -1){
                if(multipleComponent(i,v,graph,colour)== false){
                    return false;
                }
            }
        }
        return true ;  
    }
};

// using dfs
class Solution {
public:
bool dfs(int start ,int color, vector<int>&colour,vector<vector<int>>& graph){
    colour[start] = color ;
    for(auto it : graph[start]){
        if(colour[it] == -1){
            if(dfs(it,!color,colour,graph) == false){
                return false;
            }
        }
        else if(colour[it] == color){
            return false;
        }
    }
    return true;
}
    bool isBipartite(vector<vector<int>>& graph) {
        int v = graph.size();
        vector<int>colour(v,-1);
        for(int i = 0 ; i < v ; i++){
            if(colour[i] == -1) {
                if(dfs(i,0,colour,graph)==false){
                    return false;
                }
            } 
        }
        return true;
    }
};
