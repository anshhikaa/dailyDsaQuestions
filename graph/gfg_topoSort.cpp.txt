//problem : topological sort
//platofrm : gfg
// tc = o(v+e) , sc =o(v)
class Solution {
  public:
  void dfs(int node, vector<int>&vis,vector<vector<int>>&adjLs,stack<int>&st){
      vis[node] = 1 ;
      for(auto it : adjLs[node]){
          if(!vis[it]){
              dfs(it,vis,adjLs,st);
          }
      }
      st.push(node);
  }
    vector<int> topoSort(int V, vector<vector<int>>& edges) {
        // code here
        ///edges to adjLs 
        vector<vector<int>>adjLs(V);
        for(int i = 0  ; i < edges.size();i++){
            int u = edges[i][0];
            int v = edges[i][1];
            adjLs[u].push_back(v);
        }
        vector<int>vis(V,0);
        stack<int>st ;
        vector<int>ans ;
        for(int i = 0 ; i<V;i++){
            if(!vis[i]){
                dfs(i,vis,adjLs,st);
            }
        }
        while(!st.empty()){
            
            ans.push_back(st.top());
            st.pop();
        }
        return ans;
    }
};

//approach - bfs kahns algo
class Solution {
  public:
    vector<int> topoSort(int V, vector<vector<int>>& edges) {
        // code here
        queue<int>q ;
        vector<int>inDegree(V,0);
        vector<vector<int>>adjls(V);
        for(int i = 0 ; i<edges.size();i++){
            int u = edges[i][0];
            int v = edges[i][1];
            adjls[u].push_back(v);
        }
        for(int i = 0 ; i<V;i++){
             for(auto it : adjls[i]){
                 inDegree[it]++;
             }
        }
        for(int i = 0 ;i < V; i++){
            if(inDegree[i]==0){
                q.push(i);
            }
        }
        vector<int>ans;
        //bfs algo
        while(!q.empty()){
            int node = q.front(); q.pop();
            ans.push_back(node);
            for(auto it : adjls[node]){
                inDegree[it]--;
                if(inDegree[it]==0){
                    q.push(it);
                }
            }
        }
      return ans;
    }
};
