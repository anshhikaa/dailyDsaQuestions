// problem : cycle in undirected graph 
// platform gfg 
// tc = O(V+E) , sc = O(V+E)
class Solution {
  public:
  bool dfs(int node , int parent , vector<int>&vis ,vector<vector<int>>&adj){
      vis[node] = 1 ;
      for(auto it : adj[node]){
          if(!vis[it]){
              if(dfs(it,node,vis,adj) == true){
                  return true;
              }
          }
          else if(it != parent){
              //cycle detected
              return true;
          }
      }
      return false;
  }
    bool isCycle(int V, vector<vector<int>>& edges) {
        vector<int>vis(V,0);
        // Code here
       //convert edge list to adj list
       vector<vector<int>>adj(V);
       for(auto &e : edges){
           int u = e[0];
           int v =e[1];
           adj[u].push_back(v);
           adj[v].push_back(u);
       }
       //as multiple components are present
       for(int i = 0 ;  i< V ;i++){
           if(!vis[i]){
               if(dfs(i,-1,vis,adj)== true){
                   return true;
               }
           }
       }
        return false;
    }
};
