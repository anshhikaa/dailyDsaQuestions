//problem : shortest path in directed acyclic graph 
// platofrm : gfg
// approach : topo sort(dfs) + relax edges 
// tc = o(v+e) , sc = o(v)
class Solution {
  public:
  void dfs(int node , vector<int>&vis,vector<vector<pair<int,int>>>&adjls,stack<int>&st){
      vis[node] = 1 ;
      for(auto it : adjls[node]){
          int adjNode  = it.first ;
          if(!vis[adjNode]){
              dfs(adjNode,vis,adjls,st);
          }
      }
      st.push(node);
  }
    vector<int> shortestPath(int V, int E, vector<vector<int>>& edges) {
        // code here
        vector<vector<pair<int,int>>>adjls(V);
        //edges to adjls 
        for(int i = 0 ; i<E;i++){
            int u = edges[i][0];
            int v= edges[i][1];
            int c= edges[i][2];
            adjls[u].push_back({v,c});
        }
        vector<int>vis(V,0);
        stack<int>st;
        for(int i = 0 ; i<V;i++){
            if(!vis[i]){
                dfs(i,vis,adjls,st);
            }
        }
        //topo sort ->relax wt
        vector<int>ans(V) ;
        ans[0] = 0 ;
     for(int i = 1 ; i<V;i++){
         ans[i] =INT_MAX ;
     }
        while(!st.empty()){
            int top = st.top() ; st.pop();
               if(ans[top] !=INT_MAX){
               for(auto it : adjls[top]){
               int adjNode = it.first ;
               int cost = it.second;
                 ans[adjNode] = min(ans[adjNode], ans[top]+cost);
               }
           }
        }
        for(int i = 0 ; i<V;i++){
            if(ans[i] == INT_MAX) ans[i] = -1 ;
        }
        return ans ;
    }
};
