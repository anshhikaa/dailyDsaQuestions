//dijstraa algo with a two arrays (dst and ways) ways update according to visits
class Solution {
public:
    int countPaths(int n, vector<vector<int>>& roads) {
        vector<vector<pair<int,int>>>adjls(n);
        for(int i  = 0 ;i < roads.size() ;i++){
            int u = roads[i][0];
            int v = roads[i][1];
            int d = roads[i][2];
            adjls[u].push_back({v,d});
            adjls[v].push_back({u,d});
        }
        priority_queue<pair<long long,int>,vector<pair<long long,int>>,greater<pair<long long,int>>>pq ;
        vector<long long >dist(n,LLONG_MAX);
        vector<int>ways(n,0);
        dist[0]= 0 ;
        ways[0] = 1;
        pq.push({0,0});  // dist , node
        int mod = 1e9  + 7;
        while(!pq.empty()){
            auto[dis,node] = pq.top(); pq.pop();
            if(dis > dist[node]) continue;
            for(auto it : adjls[node]){
                   int nextN = it.first;
                   long long ndis = it.second;
                   if(ndis+dis < dist[nextN]){
                     dist[nextN]= ndis + dis ;
                     pq.push({ndis+dis , nextN});
                     //update stop for first time 
                     ways[nextN] = ways[node]; 
                    } 
                    else if(dis + ndis == dist[nextN]){
                        //already exploring this path 
                        ways[nextN] = (ways[node] + ways[nextN]) % mod;
                    }
                }
            }
            return ways[n-1] % mod ;
        }
};
