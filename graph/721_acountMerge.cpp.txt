//disjoint set creation , hashing , grouping , then merge
class DisjointSet{
    vector<int> rank , parent,size  ;
public :
    DisjointSet(int n ){
        //attributes
        rank.resize(n+1,0);
        parent.resize(n+1);
        size.resize(n+1);
        for(int i = 0 ; i <= n ;i++){
            parent[i]=i;
            size[i] = 1;
        }
    }
    //methods
    int findParent(int node){
        //base case
        if(parent[node]==node) return node ;
        //recursion
        return parent[node]=findParent(parent[node]);
    }
    void unionByRank(int u , int v){
        //find ultimate parent
        int ulp_u = findParent(u);
        int ulp_v = findParent(v);
        //check rank(ultimate parent)
        if(ulp_u == ulp_v) return ;
        //connect smaller to the higher rank ultimate parent
        if(rank[ulp_u] < rank[ulp_v]){
            parent[ulp_u] = ulp_v;
        }
        else if(rank[ulp_v] < rank[ulp_u]){
            parent[ulp_v] = ulp_u;
        }
        else{
            parent[ulp_v] = ulp_u;
             rank[ulp_u]++;
        }
    }
     void unionBySize(int u , int v){
        //find ultimate parent
        int ulp_u = findParent(u);
        int ulp_v = findParent(v);
        //check size(ultimate parent)
        if(ulp_u == ulp_v) return ;
        if(size[ulp_u]<size[ulp_v]){
            parent[ulp_u] = ulp_v;
            size[ulp_v] +=size[ulp_u];
        }
        else{
            parent[ulp_v]=ulp_u;
            size[ulp_u] += size[ulp_v];
        }
    }
};

class Solution {
public:
    vector<vector<string>> accountsMerge(vector<vector<string>>& accounts) {
        int n = accounts.size();
        DisjointSet ds(n);
        unordered_map<string,int>mapMailNode;
       for(int i = 0 ; i < n ;i++){
          for(int j = 1 ; j < accounts[i].size() ;j++){
                string s = accounts[i][j];
                if(mapMailNode.find(s) == mapMailNode.end()){
                mapMailNode[s]= i;
                }
                else{
                  ds.unionBySize(i, mapMailNode[s]);
                }
            }
        }
        vector<vector<string>>mergeAcc(n) ;
        for(auto it : mapMailNode){
            string mail = it.first ;
            int node = ds.findParent(it.second);
            mergeAcc[node].push_back(mail);
        }
        vector<vector<string>>ans ;
        for(int i = 0 ; i < n ; i ++){
            if(mergeAcc[i].size()==0)continue ;
            sort(mergeAcc[i].begin(),mergeAcc[i].end());
            vector<string>temp ; 
            temp.push_back(accounts[i][0]);
            for(auto it : mergeAcc[i]){
                temp.push_back(it);
            }
              ans.push_back(temp);
        } 
        return ans ; 
    }
};
