//floyd warshall
class Solution {
public:
    int findTheCity(int n, vector<vector<int>>& edges, int distanceThreshold) {
        vector<vector<int>>adjmat(n,vector<int>(n,1e8));
        for(auto it : edges){
            int u = it[0];
            int v = it[1];
            int wt = it[2] ;
            adjmat[u][v]=wt;
            adjmat[v][u]=wt;
        }
        for(int i = 0 ; i <n ;i++){
            adjmat[i][i] = 0 ;
        }
        //floyd warshall
        for(int via= 0 ; via < n ; via++){
            for(int i = 0 ; i < n ; i++){
                for(int j = 0 ; j < n ; j++){
                    if(adjmat[i][via] ==1e8 || adjmat[via][j] == 1e8) continue ;
                    adjmat[i][j] = min(adjmat[i][j] , adjmat[i][via]+adjmat[via][j]);
                }
            }
        }
        int maxCity = n ; 
        int city = -1;
        for(int i = 0 ; i < n ; i++){
            int count  = 0 ;
            for(int j = 0 ; j < n ; j++){
                if(adjmat[i][j] <= distanceThreshold){
                 count++;
                }
            }
            if (count <= maxCity){
                maxCity = count ;
                city = i ;
            }
        }
        return city ;
    }
};
