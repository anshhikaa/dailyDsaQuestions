//implementing dijkstra algo using minheap(priority queue) - tc = o(ElogV)
class Solution {
  public:
    vector<int> dijkstra(int V, vector<vector<int>> &edges, int src) {
        //edges to adjacency list 
        vector<vector<pair<int,int>>>adjls(V) ;
        for(int i = 0 ; i< edges.size();i++){
            int u = edges[i][0];
            int v = edges[i][1];
            int wt = edges[i][2];
            adjls[u].push_back({v,wt});
            adjls[v].push_back({u,wt});
        }
        priority_queue<pair<int,int>,vector<pair<int,int>>,greater<pair<int,int>>>minHeap;
        vector<int>dist(V);
        for(int i  = 0 ; i< V; i++){
            dist[i]= INT_MAX;
        }
        dist[src]= 0 ;
        minHeap.push({0,src});
        while(!minHeap.empty()){
            // int dis = minHeap.top().first;
            // int node = minHeap.top().second;
            auto[dis,node] = minHeap.top();
            minHeap.pop();
            for(auto it : adjls[node]){
                int adjNode = it.first; 
                int wt = it.second ;
                if(dis + wt < dist[adjNode] ){
                    dist[adjNode] = dis + wt ;
                    minHeap.push({dist[adjNode],adjNode});
                }
            }
        }
        return dist ;
    }
};

//using set data structure
class Solution {
  public:
    vector<int> dijkstra(int V, vector<vector<int>> &edges, int src) {
        //edges to adjacency list 
        vector<vector<pair<int,int>>>adjls(V) ;
        for(int i = 0 ; i< edges.size();i++){
            int u = edges[i][0];
            int v = edges[i][1];
            int wt = edges[i][2];
            adjls[u].push_back({v,wt});
            adjls[v].push_back({u,wt});
        }
        set<pair<int,int>>st;
        vector<int>dist(V);
        for(int i  = 0 ; i< V; i++){
            dist[i]= INT_MAX;
        }
       dist[src]= 0 ;
       st.insert({0,src});
        while(!st.empty()){
           auto it = *(st.begin());
           int dis = it.first;
           int node = it.second ;
           st.erase(it);
            for(auto it : adjls[node]){
                int adjNode = it.first; 
                int wt = it.second ;
                if(dis + wt < dist[adjNode] ){
                   if(dist[adjNode] != INT_MAX){
                       st.erase({dist[adjNode],adjNode});
                   }
                       dist[adjNode] = dis + wt;
                       st.insert({dist[adjNode],adjNode});
                }
            }
        }
        return dist ;
    }
};
