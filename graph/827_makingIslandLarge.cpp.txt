//using dsu as dynamic graph , main point while calculationg use set to store ultimate parent to avoid redundat additon of same component
class DisjointSet{
    public:
    vector<int> rank , parent,size  ;
    DisjointSet(int n ){
        //attributes
        rank.resize(n+1,0);
        parent.resize(n+1);
        size.resize(n+1);
        for(int i = 0 ; i <= n ;i++){
            parent[i]=i;
            size[i] = 1;
        }
    }
    //methods
    int findParent(int node){
        //base case
        if(parent[node]==node) return node ;
        //recursion
        return parent[node]=findParent(parent[node]);
    }
    void unionByRank(int u , int v){
        //find ultimate parent
        int ulp_u = findParent(u);
        int ulp_v = findParent(v);
        //check rank(ultimate parent)
        if(ulp_u == ulp_v) return ;
        //connect smaller to the higher rank ultimate parent
        if(rank[ulp_u] < rank[ulp_v]){
            parent[ulp_u] = ulp_v;
        }
        else if(rank[ulp_v] < rank[ulp_u]){
            parent[ulp_v] = ulp_u;
        }
        else{
            parent[ulp_v] = ulp_u;
             rank[ulp_u]++;
        }
    }
     void unionBySize(int u , int v){
        //find ultimate parent
        int ulp_u = findParent(u);
        int ulp_v = findParent(v);
        //check size(ultimate parent)
        if(ulp_u == ulp_v) return ;
        if(size[ulp_u]<size[ulp_v]){
            parent[ulp_u] = ulp_v;
            size[ulp_v] +=size[ulp_u];
        }
        else{
            parent[ulp_v]=ulp_u;
            size[ulp_u] += size[ulp_v];
        }
    }
};        

class Solution {
public:
    int largestIsland(vector<vector<int>>& grid) {
        int n = grid.size();
        DisjointSet ds(n*n);
        // step 1 connect components by union by size to get the size later on of the components
        for(int i = 0 ; i < n ; i++){
            for(int j = 0 ; j < n ; j++){
                if(grid[i][j]==0)continue ;
                //if 1 then four direction check
                int dr[]= {-1,0,1,0};
                int dc[] = {0,-1,0,1};
                for(int k = 0 ; k < 4 ; k++){
                   int ni = i + dr[k];
                   int nj = j+ dc[k];
                   if(ni >= 0 && nj >= 0 && ni < n && nj < n){
                        if(grid[ni][nj]==1){
                            int node = i*n + j ;
                            int adjNode = ni*n + nj;
                            if(ds.findParent(node) != ds.findParent(adjNode)){
                            ds.unionBySize(node,adjNode);
                           }
                       }
                   }
                }
            }
        }
        //step 2 : tranform each 0-> 1 and find ultimate parent of component store in set and extract size 
        int anssize = 0 ;
        for(int i =  0 ; i < n ;i++){
            for(int j = 0 ; j < n  ; j++){
                if(grid[i][j]==1)continue;
                int dr[]= {-1,0,1,0};
                int dc[] = {0,-1,0,1};
                set<int>st ;
                for(int k = 0 ; k < 4 ; k++){
                   int ni = i + dr[k];
                   int nj = j+ dc[k];
                   if(ni >= 0 && nj >= 0 && ni < n && nj < n){
                        if(grid[ni][nj]==1){
                           int node = ni*n + nj ;
                           int parent = ds.findParent(node);
                           st.insert(parent);
                       }
                    }
                }
                int size = 0 ;
                for(int it : st){
                    int nodesize = ds.size[it];
                    size+= nodesize ;
                }
                anssize = max(size+1,anssize);
            }
        }
        //what if all are ones then ultimate parent needs to be calculated
        for(int nodeNo = 0 ; nodeNo < n*n ; nodeNo ++){
            anssize = max(anssize, ds.size[ds.findParent(nodeNo)]);
        }
        return anssize ;
    }
};
