//two approach - correct is dijkstras , did floyd warshall for practice
class Solution {
public:
    int networkDelayTime(vector<vector<int>>& times, int n, int k) {
         vector<vector<pair<int,int>>>adjls(n+1);
        for(auto it : times){
        int u = it[0];
        int v= it[1];
        int wt = it[2];
        adjls[u].push_back({v,wt});
        }
      vector<int>dist(n+1,INT_MAX);
      dist[k] = 0;
      priority_queue<pair<int,int>,vector<pair<int,int>>,greater<pair<int,int>>>pq ;
      pq.push({0,k});
      while(!pq.empty()){
        auto[dis,node]= pq.top(); pq.pop();
        for(auto it : adjls[node]){
            int adjnode = it.first ;
            int ndis = it.second ;
            if(ndis+dis < dist[adjnode]){
                pq.push({ndis+dis,adjnode});
                dist[adjnode] = ndis+dis;
            }
        }
      }
      int ans = -1;
    for(int i = 1 ; i< n+1;i++){
    if(dist[i]==INT_MAX)return -1;
      ans= max(ans,dist[i]);
    }
    return ans ;
    }
};


class Solution {
public:
    int networkDelayTime(vector<vector<int>>& times, int n, int k) {
      //floyd warshall
      vector<vector<int>>adjmat(n+1,vector<int>(n+1,1e8));
      for(auto it : times){
        int u = it[0];
        int v= it[1];
        int wt = it[2];
        adjmat[u][v]= wt;
      }
      for(int i = 1 ; i <= n ; i++){
        adjmat[i][i]= 0 ;
      }
      for(int via = 1 ; via <= n ; via ++ ){
        for(int i = 1 ; i <= n ;i++){
            for(int j = 1 ;j <= n ;j++){
            if(adjmat[i][via] == 1e8 || adjmat[via][j] == 1e8)continue;
            adjmat[i][j]= min(adjmat[i][via]+adjmat[via][j], adjmat[i][j]);
            }
        }
      }  
      int Time = -1 ;
      for(int i = 1 ; i <= n;i++){
        if(adjmat[k][i] == 1e8) return -1;
        Time = max(Time,adjmat[k][i]);
      }
      return Time ;
    }
};
