//problem : detect cycle in directed graph 
// platform : gfg
// tc = O(V+E) , sc = o(V)
class Solution {
  public:
  bool dfs(int node , vector<int>&vis,vector<vector<int>> &adjLs,vector<int>&pathVis){
      vis[node] = 1 ;
      pathVis[node]  = 1 ;
      for(auto it : adjLs[node]){
          if(!vis[it]){
              if(dfs(it,vis,adjLs,pathVis) == true){
                  return true;
              }
          }
          else if( pathVis[it] == 1){
              return true;
          }
      }
      pathVis[node] = 0 ;
      return false;
  }
    bool isCyclic(int V, vector<vector<int>> &edges) {
        // code here
        vector<vector<int>>adjLs(V);
        for(int i = 0 ; i < edges.size() ;i++){
            int v= edges[i][0];
            int u = edges[i][1];
            adjLs[u].push_back(v);

            
        }
      vector<int>vis(V,0);
      vector<int>pathVis(V,0);
      for(int i = 0 ; i< V ; i++){
          if(!vis[i]){
              if(dfs(i,vis,adjLs,pathVis)== true){
                  return true;
              }
          }
      }
      return false;
      
    }
};


//kahs algo
class Solution {
public:
    vector<int> eventualSafeNodes(vector<vector<int>>& graph) {
        //reverse graph 
        queue<int>q ;
        int v = graph.size();
         vector<int>outDegree(v);
         vector<vector<int>>revG(v);
         for(int i = 0 ; i<v;i++){
            outDegree[i] = graph[i].size();
            for(auto it : graph[i] ){
                revG[it].push_back(i);
            }
         }
         for(int i = 0 ; i<v;i++){
            if(outDegree[i] == 0){
                q.push(i);
            }
         }
         vector<int>safe;
         while(!q.empty()){
            int node = q.front() ; q.pop();
            safe.push_back(node);
            for(auto it: revG[node]){
                outDegree[it]--;
                if(outDegree[it]==0){
                    q.push(it);
                }
            }
         }
         sort(safe.begin(),safe.end());
       return safe; 
    }
};
