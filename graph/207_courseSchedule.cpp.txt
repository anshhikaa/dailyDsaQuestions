//problem : course schedule 
// platform : leetcode 207
/ tc = o(v+e) , sc =o(v)
class Solution {
public:
    bool canFinish(int numCourses, vector<vector<int>>& prerequisites) {
        //create directed acyclic graph fron prerequisites
        vector<vector<int>>adjls(numCourses); 

        for(int i = 0 ; i<prerequisites.size();i++){
            int u = prerequisites[i][0];
            int v = prerequisites[i][1];
            adjls[v].push_back(u);
        }
        // bfs traversal 
        queue<int>q ;
        vector<int>indegree(numCourses,0);
        for(int i = 0 ; i < numCourses ; i++){
               for(auto it : adjls[i]){
                indegree[it]++;
               }
        }
        for(int i = 0 ;i<numCourses;i++){
            if(indegree[i] == 0){
                q.push(i);
            }
        }
        int count = 0;
        while(!q.empty()){
            int node = q.front(); q.pop();
            count ++;
            for(auto it : adjls[node]){
                indegree[it]--;
                if(indegree[it] == 0){
                    q.push(it);
                }
            } 
        }
     if (count == numCourses) {
        return true;
        }
     return false;
    }
};
