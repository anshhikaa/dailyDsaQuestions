//uses disjoint set 
class DisjointSet{
    vector<int> rank , parent ,size  ;
public :
    DisjointSet(int n ){
        //attributes
        rank.resize(n+1,0);
        parent.resize(n+1);
        size.resize(n+1);
        for(int i = 0 ; i <= n ;i++){
            parent[i]=i;
            size[i] = 1;
        }
    }
    //methods
    int findParent(int node){
        //base case
        if(parent[node]==node) return node ;
        //recursion
        return parent[node]=findParent(parent[node]);
    }
    void unionByRank(int u , int v){
        //find ultimate parent
        int ulp_u = findParent(u);
        int ulp_v = findParent(v);
        //check rank(ultimate parent)
        if(ulp_u == ulp_v) return ;
        //connect smaller to the higher rank ultimate parent
        else if(rank[ulp_v] < rank[ulp_u]){
            parent[ulp_u] = ulp_u;
        }
        else{
            parent[ulp_v] = ulp_u ;
             rank[ulp_u]++;
        }
    }
     void unionBySize(int u , int v){
        //find ultimate parent
        int ulp_u = findParent(u);
        int ulp_v = findParent(v);
        //check size(ultimate parent)
        if(ulp_u == ulp_v) return ;
        if(size[ulp_u]<size[ulp_v]){
            parent[ulp_u] = ulp_v;
            size[ulp_v] +=size[ulp_u];
        }
        else{
            parent[ulp_v]=ulp_u;
            size[ulp_u] += size[ulp_v];
        }
    }
};
class Solution {
  public:
  
    int kruskalsMST(int V, vector<vector<int>> &edges) {
        // code here
         sort(edges.begin(),edges.end(),[](const vector<int>a ,const vector<int>b){
             return a[2]<b[2];
         });
         int mstwt = 0 ;
         DisjointSet ds(V);
         //not same ds 
         for(auto it : edges){
             int u = it[0];
             int v= it[1];
             int wt = it[2];
             if(ds.findParent(u) != ds.findParent(v)){
                mstwt += wt;
                ds.unionBySize(u,v);
             }
         }
        return mstwt;
    }
};
