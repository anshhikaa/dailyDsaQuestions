// problem : no. of island 
//platform : leetcode 200
// tc = o(n2) , sc = o(n2)(vis array) + o(n2)(queue)
class Solution {
public:
 void bfs(int i , int j , vector<vector<char>> &grid ,vector<vector<int>>&vis){
         queue<pair<int,int>> q ;
         int n = grid.size();
         int m = grid[0].size();
         q.push({i,j});
         vis[i][j] = 1 ;
         int delRow[]  = {-1,0,1,0};
         int delCol[] =  {0,-1,0,1};
         while(!q.empty()){
             int row = q.front().first;
             int col = q.front().second;
             q.pop();
             //traverse in all 8 directions
                for(int k= 0; k < 4 ; k++){
                    int nrow = row + delRow[k];
                    int mcol = col + delCol[k];
                    if(nrow >= 0 && nrow < n && mcol >= 0 && mcol< m && !vis[nrow][mcol] && grid[nrow][mcol] == '1'){
                        vis[nrow][mcol]= 1;
                        q.push({nrow,mcol});
                    }
                }
         }   
}
    int numIslands(vector<vector<char>>& grid) {
        int n = grid.size();
        int m = grid[0].size();
        vector<vector<int>>vis(n,vector<int>(m,0));
        int count = 0 ;
        for(int i = 0 ; i <n ; i++){
            for(int j = 0 ; j < m ; j++){
                if(grid[i][j] == '1' && !vis[i][j]){
                    bfs(i , j , grid , vis );
                    count ++;
                }
            }
        }
        return count ;   
    }
};
