//problem : no. of enclaves
// platform: leetcode 1020
// tc = o(n*m) , sc = o(n*m)
class Solution {
public:
void dfs(int r  ,int c ,vector<vector<int>>&vis,vector<vector<int>>& grid){
    vis[r][c] = 1 ;
    int n = grid.size();
    int m = grid[0].size();
    int delRow[] = {0,-1,0,1};
    int delCol[] = {-1,0,1,0};
    for(int i  = 0  ; i < 4 ;i++){
        int nrow = r + delRow[i];
        int mcol = c+ delCol[i];
        if(nrow >= 0 && mcol >= 0 && nrow < n && mcol < m && !vis[nrow][mcol] && grid[nrow][mcol] == 1){
            dfs(nrow,mcol,vis,grid);
        } 
    }
}
    int numEnclaves(vector<vector<int>>& grid) {
        int n = grid.size();
        int m  =grid[0].size();
        vector<vector<int>>vis(n,vector<int>(m,0));
        //last row & first row
        for(int j = 0  ; j < m ; j++){
            if(!vis[0][j] && grid[0][j] == 1){
                dfs(0,j,vis,grid);
            }
            if(!vis[n-1][j] && grid[n-1][j]){
                dfs(n-1,j,vis,grid);
            }
        }
        // last col & first col
        for(int i = 0 ; i < n ; i++){
            if(!vis[i][0] && grid[i][0]){
                dfs(i,0,vis,grid);
            }
            if(!vis[i][m-1] && grid[i][m-1]){
                dfs(i,m-1,vis,grid);
            }
        }
        // count the inside land
        int count = 0 ; 
        for(int i = 1 ; i < n ;i++){
            for(int j = 1;j < m ;j++ ){
                if(!vis[i][j] && grid[i][j] == 1){
                    count ++;
                }
            }
        }
        return count;
    }
};

//sc = o(1) , tc = o(n*m)
class Solution {
public:
void dfs(int r  ,int c ,vector<vector<int>>& grid){
    grid[r][c]  = 0 ;
    int n = grid.size();
    int m = grid[0].size();
    int delRow[] = {0,-1,0,1};
    int delCol[] = {-1,0,1,0};
    for(int i  = 0  ; i < 4 ;i++){
        int nrow = r + delRow[i];
        int mcol = c+ delCol[i];
        if(nrow >= 0 && mcol >= 0 && nrow < n && mcol < m && grid[nrow][mcol] == 1){
            dfs(nrow,mcol,grid);
        } 
    }
}
    int numEnclaves(vector<vector<int>>& grid) {
        int n = grid.size();
        int m  =grid[0].size();
        //last row & first row
        for(int j = 0  ; j < m ; j++){
            if(grid[0][j] == 1){
                dfs(0,j,grid);
            }
            if(grid[n-1][j]){
                dfs(n-1,j,grid);
            }
        }
        // last col & first col
        for(int i = 0 ; i < n ; i++){
            if(grid[i][0]){
                dfs(i,0,grid);
            }
            if(grid[i][m-1]){
                dfs(i,m-1,grid);
            }
        }
        // count the inside land
        int count = 0 ; 
        for(int i = 1 ; i < n ;i++){
            for(int j = 1;j < m ;j++ ){
                if(grid[i][j] == 1){
                    count ++;
                }
            }
        }
        return count;
    }
};
