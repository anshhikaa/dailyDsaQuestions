//problem ; eventual safe node
// platform leetcode 802
// tc = o(v+e), sc = o(v) - only dfs - also dfs+check more better in tc
class Solution {
public:
bool dfs(int node , vector<int>&vis,vector<int>&pathVis,vector<vector<int>>& graph){
    vis[node] =1;
    pathVis[node]= 1;
    for(auto it : graph[node]){
        if(!vis[it]){
            if(dfs(it,vis,pathVis,graph)== true){
                return true;
            }
        }
        else if(pathVis[it]){
            return true;
        }
    }
    pathVis[node] = 0;
    return false;
}
    vector<int> eventualSafeNodes(vector<vector<int>>& graph) {
        int v = graph.size();
        vector<int>vis(v,0);
        vector<int>pathVis(v,0);
        vector<int>ans;
        for(int i = 0 ; i< v ;i++){
               if(dfs(i,vis,pathVis,graph) == false){
                ans.push_back(i); 
            }
        }
        return ans;
    }
};

//topo sort method o(v+e)
class Solution {
public:
    vector<int> eventualSafeNodes(vector<vector<int>>& graph) {
        //reverse graph 
        queue<int>q ;
        int v = graph.size();
         vector<int>outDegree(v);
         vector<vector<int>>revG(v);
         for(int i = 0 ; i<v;i++){
            outDegree[i] = graph[i].size();
            for(auto it : graph[i] ){
                revG[it].push_back(i);
            }
         }
         for(int i = 0 ; i<v;i++){
            if(outDegree[i] == 0){
                q.push(i);
            }
         }
         vector<int>safe;
         while(!q.empty()){
            int node = q.front() ; q.pop();
            safe.push_back(node);
            for(auto it: revG[node]){
                outDegree[it]--;
                if(outDegree[it]==0){
                    q.push(it);
                }
            }
         }
         sort(safe.begin(),safe.end());
       return safe; 
    }
};
