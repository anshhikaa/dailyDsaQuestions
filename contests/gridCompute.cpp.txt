//problem : grid compute 
// n^2 is the optimised solution 
string gridChallenge(vector<string> grid) {
   
    int n = grid.size();
    int m = grid[0].size();
    //rearrage rows - sort
    for(int i = 0 ; i < n ;i++){
        sort(grid[i].begin(),grid[i].end());
     }
    //check col 
    for(int col = 0 ;col< m;col++){
       for(int row = 1; row < n ; row++){
        if(grid[row][col] < grid[row-1][col]){
       return "NO";
        }
      }
    }
    return "YES";
}
