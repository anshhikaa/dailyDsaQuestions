long marcsCakewalk(vector<int> calorie) {
  sort(calorie.begin(),calorie.end(),[](int a , int b){
                return a >  b ;
    });
      long long power = 1 ;
    long long  miles = 0 ;
    for(int i = 0 ; i<calorie.size();i++){
        miles += power * calorie[i];
        power *= 2 ;    
   }
    return miles ;
}
